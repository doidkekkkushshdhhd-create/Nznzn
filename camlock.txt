local Players      = game:GetService("Players")
local RunService   = game:GetService("RunService")
local StarterGui   = game:GetService("StarterGui")

local LocalPlayer  = Players.LocalPlayer
local Camera       = workspace.CurrentCamera
local Mouse        = LocalPlayer:GetMouse()

local TOGGLE_KEY   = "q"
local SMOOTH       = 0.1768521
local OFFSET_X     = 0.11107611
local OFFSET_Y     = 0.11034856

local isActive     = false
local lockedTarget = nil

local function FindNearestEnemy()
    local screenCenter  = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local closestDist   = math.huge
    local closestPart   = nil

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end

        local char = player.Character
        if not char then continue end

        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")

        if not hrp or not hum or hum.Health <= 0 then continue end

        local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
        if not onScreen then continue end

        local dist = (screenCenter - Vector2.new(screenPos.X, screenPos.Y)).Magnitude
        if dist < closestDist then
            closestDist = dist
            closestPart = hrp
        end
    end

    return closestPart
end

Instance.new("UICorner", fovCircle).CornerRadius = UDim.new(1, 0)

local function enableCamlock()
    lockedTarget = FindNearestEnemy()
    if not lockedTarget then return end

    isActive                 = true
    fovCircle.Visible        = true
    toggleBtn.Text           = "Camlock: ON"
    toggleBtn.BackgroundColor3 = Color3.fromRGB(0, 140, 50)
end

local function disableCamlock()
    isActive                 = false
    lockedTarget             = nil
    fovCircle.Visible        = false
    toggleBtn.Text           = "Camlock: OFF"
    toggleBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
end

local function toggleCamlock()
    if isActive then
        disableCamlock()
    else
        enableCamlock()
    end
end

RunService.Heartbeat:Connect(function()
    if not isActive then return end
    if not lockedTarget or not lockedTarget.Parent then
        disableCamlock()
        return
    end

    local aimPos = lockedTarget.Position
        + lockedTarget.Velocity * SMOOTH
        + Vector3.new(OFFSET_X, OFFSET_Y, 0)

    Camera.CFrame = CFrame.new(Camera.CFrame.p, aimPos)
end)

toggleBtn.MouseButton1Click:Connect(toggleCamlock)

Mouse.KeyDown:Connect(function(pressedKey)
    if pressedKey == TOGGLE_KEY then
        toggleCamlock()
    end
end)

LocalPlayer.CharacterAdded:Connect(function()
    disableCamlock()
end)
